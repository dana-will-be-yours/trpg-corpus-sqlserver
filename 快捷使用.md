## 查劇情事件

FROM dbo.Plot_Event

## 查逐字稿發言
SELECT 
    session_id,
    scene_id,
    turn_no,
    speaker_type,
    utterance_text,
    utterance_function,
    is_in_character
FROM dbo.Utterance
ORDER BY session_id, turn_no;
## 查 GM 旁白

FROM dbo.GM_Narration;
## 查決策紀錄

FROM dbo.Decision_Log;
## 查 SMM / TMS 題項
SELECT 
    c.construct_code,
    c.construct_name,
    i.item_code,
    i.item_text
FROM dbo.Scale_Construct c
JOIN dbo.Scale_Item i
    ON c.construct_id = i.construct_id
ORDER BY c.construct_code, i.item_code;
