# 快捷使用

以下查詢依目前 `database/00_create_database.sql` 至 `database/38_dago_runtime_bundle.sql` 撰寫，可直接貼到 SSMS 執行。

```sql
USE TRPG_Corpus_DB;
GO
```

## 查資料庫物件

```sql
SELECT s.name AS schema_name, t.name AS table_name
FROM sys.tables AS t
INNER JOIN sys.schemas AS s
    ON s.schema_id = t.schema_id
WHERE s.name IN (N'dbo', N'stg')
ORDER BY s.name, t.name;

SELECT s.name AS schema_name, p.name AS procedure_name
FROM sys.procedures AS p
INNER JOIN sys.schemas AS s
    ON s.schema_id = p.schema_id
WHERE s.name IN (N'dbo', N'stg')
ORDER BY s.name, p.name;

SELECT s.name AS schema_name, v.name AS view_name
FROM sys.views AS v
INNER JOIN sys.schemas AS s
    ON s.schema_id = v.schema_id
WHERE s.name IN (N'dbo', N'stg')
ORDER BY s.name, v.name;
```

## 查研究專案與隊伍

```sql
SELECT project_id, project_code, project_title, research_design, data_status
```

## 查 PC 與 NPC

```sql
SELECT
    t.team_code,
    tm.member_code,
    pc.character_id,
    pc.character_code,
    pc.character_name,
    pc.archetype,
    pc.character_status
FROM dbo.Player_Character AS pc
INNER JOIN dbo.Team_Member AS tm
    ON tm.team_member_id = pc.team_member_id
INNER JOIN dbo.Team AS t
    ON t.team_id = tm.team_id
ORDER BY t.team_code, pc.character_code;

SELECT
    t.team_code,
    npc.npc_id,
    npc.npc_code,
    npc.npc_name,
    npc.npc_type,
    npc.role_in_story,
    npc.location_name,
    npc.npc_status
FROM dbo.NPC AS npc
INNER JOIN dbo.Team AS t
    ON t.team_id = npc.team_id
ORDER BY t.team_code, npc.npc_code;
```

## 查場次、場景與逐字稿

```sql
SELECT
    t.team_code,
    s.session_id,
    s.session_code,
    s.session_no,
    s.session_title,
    s.session_type,
    s.session_date,
    s.transcript_status,
    s.session_status
FROM dbo.TRPG_Session AS s
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
ORDER BY t.team_code, s.session_no;

SELECT
    t.team_code,
    s.session_code,
    sc.scene_id,
    sc.scene_code,
    sc.scene_no,
    sc.scene_title,
    sc.scene_type,
    sc.start_turn_no,
    sc.end_turn_no,
    sc.review_status
FROM dbo.Scene AS sc
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = sc.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
ORDER BY t.team_code, s.session_no, sc.scene_no;

SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    u.turn_no,
    u.sub_turn_no,
    u.utterance_code,
    u.speaker_type,
    COALESCE(tm.member_code, pc.character_code, npc.npc_code, u.speaker_label_raw) AS speaker_code,
    u.utterance_function,
    u.is_in_character,
    u.is_gm_narration,
    u.is_rule_related,
    u.is_decision_related,
    u.is_knowledge_related,
    u.emotion_label,
    u.utterance_text_raw
FROM dbo.Utterance AS u
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = u.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = u.scene_id
LEFT JOIN dbo.Team_Member AS tm
    ON tm.team_member_id = u.speaker_member_id
LEFT JOIN dbo.Player_Character AS pc
    ON pc.character_id = u.speaker_character_id
LEFT JOIN dbo.NPC AS npc
    ON npc.npc_id = u.speaker_npc_id
ORDER BY t.team_code, s.session_no, u.turn_no, u.sub_turn_no;
```

## 查劇情事件、決策與知識查詢

```sql
SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    pe.event_no,
    pe.event_code,
    pe.event_title,
    pe.event_type,
    pe.event_function,
    pe.event_importance,
    pe.event_summary
FROM dbo.Plot_Event AS pe
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = pe.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = pe.scene_id
ORDER BY t.team_code, s.session_no, pe.event_no;

SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    d.decision_no,
    d.decision_code,
    d.decision_title,
    d.decision_type,
    d.decision_status,
    d.consensus_level,
    d.consensus_quality_score,
    d.selected_option,
    d.decision_summary
FROM dbo.Decision_Log AS d
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = d.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = d.scene_id
ORDER BY t.team_code, s.session_no, d.decision_no;

SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    k.retrieval_no,
    k.retrieval_code,
    k.retrieval_type,
    k.retrieval_purpose,
    k.query_initiator_role,
    COALESCE(tm.member_code, pc.character_code) AS initiator_code,
    k.retrieval_source_type,
    k.search_keyword,
    k.retrieval_success,
    k.retrieval_success_level,
    k.tms_role,
    k.result_used_summary
FROM dbo.Knowledge_Retrieval_Log AS k
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = k.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = k.scene_id
LEFT JOIN dbo.Team_Member AS tm
    ON tm.team_member_id = k.query_initiator_member_id
LEFT JOIN dbo.Player_Character AS pc
    ON pc.character_id = k.query_initiator_character_id
ORDER BY t.team_code, s.session_no, k.retrieval_no;
```

## 查 staging 匯入狀態

```sql
DECLARE @batch_code NVARCHAR(100) = N'TRPG_UTT_20260505_T01_S01_v01';

SELECT
    import_batch_id,
    batch_code,
    project_code,
    source_table_name,
    source_file_name,
    import_status,
    total_row_count,
    valid_row_count,
    invalid_row_count,
    warning_count,
    error_count
FROM stg.Import_Batch
WHERE batch_code = @batch_code;

SELECT
    ui.source_row_no,
    ui.project_code,
    ui.team_code,
    ui.session_code,
    ui.scene_code,
    ui.turn_no_text,
    ui.speaker_type,
    ui.speaker_code,
    ui.utterance_function,
    ui.import_status,
    ui.validation_error,
    ui.validation_warning
FROM stg.Utterance_Import AS ui
INNER JOIN stg.Import_Batch AS ib
    ON ib.import_batch_id = ui.import_batch_id
WHERE ib.batch_code = @batch_code
ORDER BY ui.source_row_no;
```
