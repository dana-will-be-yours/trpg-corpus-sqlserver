## 查資料庫物件
```sql
SELECT s.name AS schema_name, t.name AS table_name
FROM sys.tables AS t
INNER JOIN sys.schemas AS s
    ON s.schema_id = t.schema_id
WHERE s.name IN (N'dbo', N'stg')
ORDER BY s.name, t.name;

SELECT s.name AS schema_name, p.name AS procedure_name
FROM sys.procedures AS p
INNER JOIN sys.schemas AS s
    ON s.schema_id = p.schema_id
WHERE s.name = N'stg'
ORDER BY p.name;
```

## 查逐字稿發言

```sql
SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    u.turn_no,
    u.sub_turn_no,
    u.utterance_code,
    u.speaker_type,
    COALESCE(tm.member_code, pc.character_code, npc.npc_code, u.speaker_label_raw) AS speaker_code,
    u.utterance_function,
    u.is_in_character,
    u.is_gm_narration,
    u.is_rule_related,
    u.is_decision_related,
    u.is_knowledge_related,
    u.emotion_label,
    u.utterance_text_raw
FROM dbo.Utterance AS u
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = u.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = u.scene_id
LEFT JOIN dbo.Team_Member AS tm
    ON tm.team_member_id = u.speaker_member_id
LEFT JOIN dbo.Player_Character AS pc
    ON pc.character_id = u.speaker_character_id
LEFT JOIN dbo.NPC AS npc
    ON npc.npc_id = u.speaker_npc_id
ORDER BY t.team_code, s.session_no, u.turn_no, u.sub_turn_no;
```

## 查劇情事件與事件因果鏈

```sql
SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    pe.event_no,
    pe.event_code,
    pe.event_title,
    pe.event_type,
    pe.event_function,
    pe.event_importance,
    pe.event_summary
FROM dbo.Plot_Event AS pe
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = pe.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
LEFT JOIN dbo.Scene AS sc
    ON sc.scene_id = pe.scene_id
ORDER BY t.team_code, s.session_no, pe.event_no;

SELECT
    t.team_code,
    s.session_code,
    ecl.causal_link_code,
    cause.event_code AS cause_event_code,
    cause.event_title AS cause_event_title,
    effect.event_code AS effect_event_code,
    effect.event_title AS effect_event_title,
    ecl.link_type,
    ecl.link_strength,
    ecl.causal_direction
FROM dbo.Event_Causal_Link AS ecl
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = ecl.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
INNER JOIN dbo.Plot_Event AS cause
    ON cause.plot_event_id = ecl.cause_event_id
INNER JOIN dbo.Plot_Event AS effect
    ON effect.plot_event_id = ecl.effect_event_id
ORDER BY t.team_code, s.session_no, ecl.causal_link_code;
```

## 查 SMM 與 TMS 註記

```sql
SELECT
    t.team_code,
    s.session_code,
    sc.scene_code,
    sc.scene_title,
    sc.smm_note,
    sc.tms_note,
    sc.knowledge_summary,
    sc.decision_summary
FROM dbo.Scene AS sc
INNER JOIN dbo.TRPG_Session AS s
    ON s.session_id = sc.session_id
INNER JOIN dbo.Team AS t
    ON t.team_id = s.team_id
WHERE sc.smm_note IS NOT NULL
   OR sc.tms_note IS NOT NULL
   OR sc.knowledge_summary IS NOT NULL
   OR sc.decision_summary IS NOT NULL
ORDER BY t.team_code, s.session_no, sc.scene_no;

SELECT
    t.team_code,
    h.history_no,
    h.history_code,
    h.history_title,
    h.history_type,
    h.smm_state_note,
    h.smm_change_note,
    h.tms_state_note,
    h.tms_change_note,
    h.knowledge_sharing_summary,
    h.database_use_summary
FROM dbo.Team_Play_History AS h
INNER JOIN dbo.Team AS t
    ON t.team_id = h.team_id
WHERE h.smm_state_note IS NOT NULL
   OR h.smm_change_note IS NOT NULL
   OR h.tms_state_note IS NOT NULL
   OR h.tms_change_note IS NOT NULL
   OR h.knowledge_sharing_summary IS NOT NULL
   OR h.database_use_summary IS NOT NULL
ORDER BY t.team_code, h.history_no;
```

## 查 staging 匯入狀態

```sql
DECLARE @batch_code NVARCHAR(100) = N'TRPG_UTT_20260505_T01_S01_v01';

SELECT
    import_batch_id,
    batch_code,
    project_code,
    source_table_name,
    source_file_name,
    import_status,
    total_row_count,
    valid_row_count,
    invalid_row_count,
    warning_count,
    error_count
FROM stg.Import_Batch
WHERE batch_code = @batch_code;

SELECT
    ui.source_row_no,
    ui.project_code,
    ui.team_code,
    ui.session_code,
    ui.scene_code,
    ui.turn_no_text,
    ui.speaker_type,
    ui.speaker_code,
    ui.utterance_function,
    ui.import_status,
    ui.validation_error,
    ui.validation_warning
FROM stg.Utterance_Import AS ui
INNER JOIN stg.Import_Batch AS ib
    ON ib.import_batch_id = ui.import_batch_id
WHERE ib.batch_code = @batch_code
ORDER BY ui.source_row_no;
```
